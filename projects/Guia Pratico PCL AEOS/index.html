🐍 Integração Programática via Antigravity Python SDK 
import asyncio
from antigravity import AgentClient, TaskConfig

async def main():
    # Conecta ao Antigravity Core local apontando para o workspace PCL AEOS
    client = AgentClient(workspace="c:/PromptCore_Labs")
    
    # Aluga um agente especializado para executar uma tarefa de conformidade
    lease = await client.lease_agent(
        role="Lead_TLC_Engineer",
        skills=["cortex", "laya"]
    )
    
    # Dispara o objetivo autônomo com rastreamento nos transcripts
    result = await lease.execute_goal(
        prompt="Validar se todos os schemas em memory.* possuem recibos de evidência SHA-256"
    )
    print(f"Status da Auditoria: {result.status} | Recibos verificados: {result.data}")

asyncio.run(main())
